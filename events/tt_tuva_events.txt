# ----------------------------------------------------------------------
#  The Taishet Peace: Tuvan peace offer to SOV
# ----------------------------------------------------------------------
country_event = {
	id = tt_tuva.10
	title = tt_tuva.10.t
	desc = tt_tuva.10.desc
	picture = GFX_report_event_generic_sign_treaty2

	is_triggered_only = yes

	# Option A: Moscow cedes Siberia
	option = {
		name = tt_tuva.10.a

		ai_chance = {
			base = 90
		}

		every_state = {
			limit = {
				is_owned_by = SOV
				is_controlled_by = TAN
				OR = {
					state = 40
					state = 654
					state = 563
					state = 564
					state = 565
					state = 566
					state = 567
					state = 568
					state = 569
					state = 570
				}
			}
			transfer_state_to = TAN
		}

		white_peace = TAN

		TAN = {
			if = {
				limit = { has_idea = TT_siberian_gambit }
				swap_ideas = {
					remove_idea = TT_siberian_gambit
					add_idea = TT_masters_of_siberia
				}
			}
			else = {
				add_ideas = TT_masters_of_siberia
			}
			add_political_power = 150
			add_stability = 0.10
			news_event = {
				id = tt_tuva.11
				days = 1
			}
		}
	}

	# Option B: Fight to the bitter end (primarily for human SOV player)
	option = {
		name = tt_tuva.10.b

		ai_chance = {
			base = 10
			modifier = {
				factor = 0
				surrender_progress > 0.15
			}
		}

		add_stability = -0.10
		add_war_support = -0.10
	}
}
