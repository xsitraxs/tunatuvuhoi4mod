add_namespace = tt_postwar

country_event = {
	id = tt_postwar.1
	title = tt_postwar.1.t
	desc = tt_postwar.1.desc
	picture = GFX_tt_tuva_negotiations
	is_triggered_only = yes
	fire_only_once = yes
	trigger = { tag = TAN has_country_flag = TT_postwar_council_started NOT = { has_country_flag = TT_postwar_route_chosen } }
	option = {
		name = tt_postwar.1.a
		ai_chance = { base = 40 }
		if = {
			limit = { NOT = { has_country_flag = TT_postwar_route_chosen } }
			set_country_flag = TT_postwar_route_chosen
			set_country_flag = TT_postwar_route_military
			add_ideas = TT_postwar_military_base
		}
	}
	option = {
		name = tt_postwar.1.b
		ai_chance = { base = 30 }
		if = {
			limit = { NOT = { has_country_flag = TT_postwar_route_chosen } }
			set_country_flag = TT_postwar_route_chosen
			set_country_flag = TT_postwar_route_local
			add_ideas = TT_postwar_local_base
		}
	}
	option = {
		name = tt_postwar.1.c
		ai_chance = { base = 30 }
		if = {
			limit = { NOT = { has_country_flag = TT_postwar_route_chosen } }
			set_country_flag = TT_postwar_route_chosen
			set_country_flag = TT_postwar_route_modern
			add_ideas = TT_postwar_modern_base
		}
	}
}
