# Clears stale one-shot request flags daily. A response event addressed to a
# country that no longer exists is dropped by the engine, so it can never
# consume a newer request: clearing on the recipient's disappearance (or, for
# the Taishet demand, on the end of the war) is safe and keeps the "an old
# answer must not eat a newer request" invariant. No timer expirations here.
TT_cleanup_diplomatic_pending = {
	if = {
		limit = { has_country_flag = TT_taishet_request_pending NOT = { has_war_with = SOV } }
		clr_country_flag = TT_taishet_request_pending
	}
	if = {
		limit = { has_country_flag = TT_corridor_request_pending NOT = { country_exists = SOV } }
		clr_country_flag = TT_corridor_request_pending
	}
	if = {
		limit = { has_country_flag = TT_war_mandate_pending NOT = { country_exists = SOV } }
		clr_country_flag = TT_war_mandate_pending
	}
}

# SOV country scope: guaranteed package PLUS other Soviet states controlled by TAN.
# Run before white_peace, which would otherwise return the extra occupations.
TT_transfer_taishet_package = {
	if = {
		limit = { tag = SOV country_exists = TAN TAN = { TT_taishet_response_valid = yes } }
		every_state = {
			limit = {
				is_owned_by = SOV
				OR = {
					is_controlled_by = TAN
					state = 40
					state = 654
					state = 563
					state = 564
					state = 565
					state = 566
					state = 567
					state = 568
					state = 569
					state = 570
					state = 574
					state = 575
					state = 576
					state = 578
				}
			}
			transfer_state_to = TAN
		}
	}
}
